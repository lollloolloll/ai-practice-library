@@term type-safe todo-app ts
@@prompt
할 일 목록 앱을 만들어줘. 완전히 타입 안전(type-safe)하게 — TypeScript strict, `any` 금지, 외부에서 들어오는 데이터는 zod로 검증해줘.
@@transcript
타입 안전성을 세 겹으로 확보했습니다.

1. **컴파일 타임**: `strict` + `noUncheckedIndexedAccess` + ESLint `no-explicit-any`
2. **런타임 경계 검증**: localStorage·API처럼 "타입을 믿을 수 없는 곳"에서 들어온 값은 zod 스키마로 파싱 — 실패하면 타입이 `Todo`가 되지 않음
3. **타입을 스키마에서 파생**: `type Todo = z.infer<typeof TodoSchema>` — 스키마와 타입이 따로 놀 수 없음

`as` 단언은 한 군데도 쓰지 않았습니다.

export const TodoSchema = z.object({
  id: z.string().uuid(),
  text: z.string().trim().min(1, "내용을 입력하세요").max(200, "200자 이하로 입력하세요"),

export const NewTodoInputSchema = TodoSchema.pick({ text: true });

export type Todo = z.infer<typeof TodoSchema>;
export type NewTodoInput = z.infer<typeof NewTodoInputSchema>;

export type Filter = "all" | "active" | "done";

export type TodoAction =
  | { type: "add"; text: string }
  | { type: "toggle"; id: Todo["id"] }
  | { type: "remove"; id: Todo["id"] };

import { TodoListSchema, type Todo } from "./schema";

const KEY = "todos";

export type LoadResult = { ok: true; todos: Todo[] } | { ok: false; error: string };

export function loadTodos(): LoadResult {
  let raw: unknown;
  try {
    raw = JSON.parse(localStorage.getItem(KEY) ?? "[]");
  } catch {
    return { ok: false, error: "저장 데이터가 JSON이 아닙니다." };
  }
  const parsed = TodoListSchema.safeParse(raw);
  return parsed.success ? { ok: true, todos: parsed.data } : { ok: false, error: parsed.error.issues[0]?.message ?? "형식 오류" };
}

export function saveTodos(todos: readonly Todo[]): void {
  localStorage.setItem(KEY, JSON.stringify(todos));
}
@@file src/reducer.ts
import type { Todo, TodoAction } from "./schema";

export function reducer(todos: readonly Todo[], action: TodoAction): Todo[] {
  switch (action.type) {
    case "add":
      return [...todos, { id: crypto.randomUUID(), text: action.text, done: false, createdAt: Date.now() }];
    case "toggle":
      return todos.map((t) => (t.id === action.id ? { ...t, done: !t.done } : t));
    case "remove":
      return todos.filter((t) => t.id !== action.id);
    default: {
      const exhaustive: never = action; // 새 액션을 추가하고 처리하지 않으면 컴파일 오류
      return exhaustive;
    }
  }
}

import { NewTodoInputSchema, type Todo } from "./schema";
import { loadTodos, saveTodos } from "./storage";
import { reducer } from "./reducer";

function getEl<T extends Element>(selector: string, ctor: new () => T): T {
  const el = document.querySelector(selector);
  if (!(el instanceof ctor)) throw new Error(`${selector} 요소가 없거나 타입이 다릅니다`);
  return el;
}

const form = getEl("#form", HTMLFormElement);
const input = getEl("#input", HTMLInputElement);
const list = getEl("#list", HTMLUListElement);
const error = getEl("#error", HTMLParagraphElement);

const initial = loadTodos();
let todos: Todo[] = initial.ok ? initial.todos : [];
if (!initial.ok) error.textContent = initial.error;

form.addEventListener("submit", (e) => {
  e.preventDefault();
  const result = NewTodoInputSchema.safeParse({ text: input.value });
  if (!result.success) {
    error.textContent = result.error.issues[0]?.message ?? "";
    return;
  }
  error.textContent = "";
  todos = reducer(todos, { type: "add", text: result.data.text });
  saveTodos(todos);
  input.value = "";
  render();
});

function render(): void {
  list.replaceChildren(...todos.map(renderItem));
}

function renderItem(todo: Todo): HTMLLIElement {
  const li = document.createElement("li");
  const box = document.createElement("input");
  box.type = "checkbox";
  box.checked = todo.done;
  box.addEventListener("change", () => dispatchAndSave({ type: "toggle", id: todo.id }));
  const span = document.createElement("span");
  span.textContent = todo.text;
  const del = document.createElement("button");
  del.textContent = "삭제";
  del.addEventListener("click", () => dispatchAndSave({ type: "remove", id: todo.id }));
  li.append(box, span, del);
  return li;
}

function dispatchAndSave(action: Parameters<typeof reducer>[1]): void {
  todos = reducer(todos, action);
  saveTodos(todos);
  render();
}

render();

@@transcript
실패할 수 있는 지점마다 처리 방식을 정했습니다.

| 실패 지점 | 처리 |
|---|---|
| 저장 데이터 손상 (JSON 파싱 실패) | 원본을 `todos.backup`에 보존 → 빈 목록으로 시작 → 안내 |
| 저장 실패 (용량 초과·차단) | 화면 상태는 유지, 오류 배너 + **다시 시도** 버튼 |
| 예상 못 한 런타임 오류 | `window.onerror`에서 잡아 배너 표시 (흰 화면 방지) |

@@term edge-cases todo-app html
@@prompt
할 일 목록 앱을 만들어줘. 엣지 케이스(edge cases)도 처리해줘 — 공백만 입력, 아주 긴 제목, 중복 항목, 손상된 localStorage, 한글 입력 중 Enter 같은 경우.
@@transcript
처리한 엣지 케이스 목록입니다.

1. **공백만 입력** → 추가 안 함 (`trim()`), 연속 공백은 하나로
2. **긴 제목** → 최대 200자, 남은 글자 수 표시, 긴 영단어/URL은 줄바꿈 (`overflow-wrap: anywhere`)
3. **중복** → 같은 내용(대소문자·공백 무시)이 이미 진행 중이면 추가하지 않고 기존 항목을 강조
4. **손상된 localStorage** → 파싱 실패·배열 아님·필드 누락 항목은 걸러냄
5. **한글 IME 조합 중 Enter** → `e.isComposing` 체크로 마지막 글자가 두 번 들어가는 버그 방지
6. **인덱스 대신 ID** → 빠르게 연속 삭제해도 엉뚱한 항목이 지워지지 않음
7. **빈 목록 / 전부 완료** → 각각 다른 안내 문구

@@prompt
할 일 목록 앱이 가끔 열자마자 빈 화면이 되고 콘솔에 "TypeError: todos.forEach is not a function"이 떠. try/catch로 오류를 숨기지 말고 근본 원인(root cause)을 찾아서 고쳐줘.
@@transcript
**재현**: 콘솔에서 `localStorage.setItem('todos', '{"items":[]}')` 후 새로고침 → 동일한 오류 발생.

**근본 원인**: 첫 줄의 `JSON.parse(localStorage.getItem('todos')) || []`는 "값이 없을 때"만 막아줍니다. 저장된 값이 **배열이 아닌 다른 JSON**(예: 이전 버전 앱이 `{ items: [...] }` 형태로 저장한 데이터)이면 객체가 그대로 `todos`에 들어가고, `render()`의 `todos.forEach`에서 터집니다. 같은 키 `todos`를 쓰던 예전 버전이 있었을 가능성이 큽니다.

**하지 않은 것**: `render()`를 try/catch로 감싸면 오류는 사라지지만 사용자의 기존 할 일이 조용히 안 보이게 됩니다. 그래서 증상이 아니라 원인(로드 단계의 형식 불일치)을 고쳤습니다.

**수정**: `loadTodos()`에서 형식을 확인하고, 예전 형식 `{ items }`는 **마이그레이션**해서 데이터를 살립니다. 알 수 없는 형식만 빈 목록으로 시작합니다.

```diff
-  let todos = JSON.parse(localStorage.getItem('todos')) || [];
+  function loadTodos() {
+    const data = JSON.parse(localStorage.getItem('todos') || '[]');
+    if (Array.isArray(data)) return data;
+    if (data && Array.isArray(data.items)) return data.items; // 이전 버전 형식 마이그레이션
+    return [];
+  }
+  let todos = loadTodos();
+  save(); // 마이그레이션된 형식으로 다시 저장
```

const STORAGE_KEY = 'todos';

const todos = JSON.parse(localStorage.getItem(STORAGE_KEY) ?? '[]');

const save = () => localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));

function render() {
  const list = document.getElementById('todoList');
  list.replaceChildren(
    ...todos.map((todo) => {
      const li = document.createElement('li');
      li.classList.toggle('done', todo.done);
      li.innerHTML = '<input type="checkbox"><span></span><button>삭제</button>';
      li.querySelector('input').checked = todo.done;
      li.querySelector('span').textContent = todo.text;
      li.querySelector('input').addEventListener('change', () => {
        todo.done = !todo.done;
        save();
        render();
      });
      li.querySelector('button').addEventListener('click', () => {
        todos.splice(todos.indexOf(todo), 1);
        save();
        render();
      });
      return li;
    }),
  );
}

document.getElementById('form').addEventListener('submit', (event) => {
  event.preventDefault();
  const input = document.getElementById('todoInput');
  const text = input.value.trim();
  if (text === '') return;
  todos.push({ text, done: false });
  input.value = '';
  save();
  render();
});

render();
