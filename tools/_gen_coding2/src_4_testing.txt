export function addTodo(todos, rawText) {
  const text = rawText.trim();
  if (!text) return todos;
  return [...todos, { id: crypto.randomUUID(), text, done: false }];
}

export const toggleTodo = (todos, id) =>
  todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));

export const removeTodo = (todos, id) => todos.filter((t) => t.id !== id);

export const remaining = (todos) => todos.filter((t) => !t.done).length;

let todos = JSON.parse(localStorage.getItem('todos') || '[]');

function update(next) {
  todos = next;
  localStorage.setItem('todos', JSON.stringify(todos));
  render();

function render() {
  const list = document.getElementById('list');
  list.replaceChildren(...todos.map((t) => {
    const li = document.createElement('li');
    li.className = t.done ? 'done' : '';
    li.innerHTML = '<input type="checkbox"><span></span><button>삭제</button>';

    li.querySelector('span').textContent = t.text;
    li.querySelector('input').onchange = () => update(toggleTodo(todos, t.id));
    li.querySelector('button').onclick = () => update(removeTodo(todos, t.id));
    return li;
  }));

}

@@file src/todoReducer.js
export const initialState = { todos: [], filter: 'all' };

export function todoReducer(state, action) {
  switch (action.type) {
    case 'ADD': {
      const text = action.text.trim();
      if (!text) return state;
      return { ...state, todos: [...state.todos, { id: action.id, text, done: false }] };
    }
    case 'TOGGLE':
      return { ...state, todos: state.todos.map((t) => (t.id === action.id ? { ...t, done: !t.done } : t)) };
    case 'DELETE':
      return { ...state, todos: state.todos.filter((t) => t.id !== action.id) };
    case 'CLEAR_DONE':
      return { ...state, todos: state.todos.filter((t) => !t.done) };
    case 'SET_FILTER':
      return { ...state, filter: action.filter };
    default:
      return state;
  }
}

export function filterTodos(todos, filter) {
  if (filter === 'active') return todos.filter((t) => !t.done);
  if (filter === 'done') return todos.filter((t) => t.done);
  return todos;
}

export function countLabel(todos) {
  const left = todos.filter((t) => !t.done).length;
  if (todos.length === 0) return '할 일이 없습니다';
  if (left === 0) return '모두 완료!';
  return `남은 할 일 ${left}개`;
}

경계값 확인을 위해 "정확히 100자는 통과" 케이스를 하나 추가했습니다.
@@file src/validateTodo.js
export const MAX_LENGTH = 100;

/**
 * @param {string} input 사용자가 입력한 원본
 * @param {{ text: string }[]} existing 현재 할 일 목록
 * @returns {{ ok: true, value: string } | { ok: false, error: string }}
 */
export function validateTodo(input, existing = []) {
  const value = input.trim().replace(/\s+/g, ' ');
  if (!value) return { ok: false, error: '내용을 입력하세요' };
  if (value.length > MAX_LENGTH) return { ok: false, error: `${MAX_LENGTH}자 이하로 입력하세요` };
  if (existing.some((t) => t.text.trim().replace(/\s+/g, ' ') === value)) {
    return { ok: false, error: '이미 있는 할 일입니다' };
  }
  return { ok: true, value };
}
