@@code
const { useState, useEffect } = React;

const FILTERS = [['all', '전체'], ['active', '진행 중'], ['done', '완료']];

function App() {
  const [todos, setTodos] = useState(() => JSON.parse(localStorage.getItem('todos') || '[]'));
  const [text, setText] = useState('');
  const [filter, setFilter] = useState('all');

  // todos가 바뀔 때마다 localStorage에 저장
  useEffect(() => {
    localStorage.setItem('todos', JSON.stringify(todos));
  }, [todos]);

  const addTodo = (e) => {
    e.preventDefault();
    if (!text.trim()) return;
    setTodos([...todos, { id: Date.now(), text: text.trim(), done: false }]);
    setText('');
  };
  const toggle = (id) => setTodos(todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));
  const remove = (id) => setTodos(todos.filter((t) => t.id !== id));

  const visible = todos.filter((t) => (filter === 'all' ? true : filter === 'done' ? t.done : !t.done));
  const left = todos.filter((t) => !t.done).length;

  return (
    <div className="app">
      <h1>할 일 목록</h1>
      <form onSubmit={addTodo}>
        <input value={text} onChange={(e) => setText(e.target.value)} placeholder="할 일을 입력하세요" />
        <button type="submit">추가</button>
      </form>
      <div className="filters">
        {FILTERS.map(([key, label]) => (
          <button key={key} className={filter === key ? 'active' : ''} onClick={() => setFilter(key)}>
            {label}
          </button>
        ))}
      </div>
      <ul>
        {visible.map((t) => (
          <li key={t.id} className={t.done ? 'done' : ''}>
            <input type="checkbox" checked={t.done} onChange={() => toggle(t.id)} />
            <span>{t.text}</span>
            <button onClick={() => remove(t.id)}>삭제</button>
          </li>
        ))}
      </ul>
      {todos.length === 0 && <p className="empty">아직 할 일이 없어요.</p>}
      <p className="count">남은 할 일 {left}개</p>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App />);

할 일 목록 앱을 만들어줘. UI는 shadcn/ui 컴포넌트(Button, Card, Input, Checkbox)와 Tailwind로 만들어줘.
@@code
"use client";

import { useState } from "react";
import { Trash2 } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Checkbox } from "@/components/ui/checkbox";
import { Input } from "@/components/ui/input";
import { Badge } from "@/components/ui/badge";

type Todo = { id: string; text: string; done: boolean };

export default function TodoApp() {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [text, setText] = useState("");

  const add = (e: React.FormEvent) => {
    e.preventDefault();
    if (!text.trim()) return;
    setTodos((prev) => [...prev, { id: crypto.randomUUID(), text: text.trim(), done: false }]);
    setText("");
  };
  const left = todos.filter((t) => !t.done).length;

  return (
    <Card className="mx-auto mt-12 w-full max-w-md">
      <CardHeader>
        <CardTitle>할 일 목록</CardTitle>
        <CardDescription>오늘 할 일을 정리해 보세요.</CardDescription>
      </CardHeader>
      <CardContent className="space-y-4">
        <form onSubmit={add} className="flex gap-2">
          <Input value={text} onChange={(e) => setText(e.target.value)} placeholder="할 일을 입력하세요" />
          <Button type="submit">추가</Button>
        </form>
        <ul className="divide-y rounded-md border">
          {todos.map((t) => (
            <li key={t.id} className="flex items-center gap-3 px-3 py-2">
              <Checkbox
                id={t.id}
                checked={t.done}
                onCheckedChange={() => setTodos((p) => p.map((x) => (x.id === t.id ? { ...x, done: !x.done } : x)))}
              />
              <label htmlFor={t.id} className={`flex-1 text-sm ${t.done ? "text-muted-foreground line-through" : ""}`}>
                {t.text}
              </label>
              <Button variant="ghost" size="icon" onClick={() => setTodos((p) => p.filter((x) => x.id !== t.id))}>
                <Trash2 className="h-4 w-4" />
              </Button>

import type { AppState, Filter, Todo, TodoId } from "./types";

const STORAGE_KEY = "todos";

function isTodo(value: unknown): value is Todo {
  if (typeof value !== "object" || value === null) return false;
  const v = value as Record<string, unknown>;
  return typeof v.id === "string" && typeof v.text === "string" && typeof v.done === "boolean";
}

export function loadTodos(): Todo[] {
  try {
    const parsed: unknown = JSON.parse(localStorage.getItem(STORAGE_KEY) ?? "[]");
    return Array.isArray(parsed) ? parsed.filter(isTodo) : [];
  } catch {
    return [];
  }
}

export function saveTodos(todos: readonly Todo[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
}

export const addTodo = (state: AppState, text: string): AppState => ({
  ...state,
  todos: [...state.todos, { id: crypto.randomUUID(), text, done: false, createdAt: Date.now() }],
});

export const toggleTodo = (state: AppState, id: TodoId): AppState => ({
  ...state,
  todos: state.todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t)),
});

export const removeTodo = (state: AppState, id: TodoId): AppState => ({
  ...state,
  todos: state.todos.filter((t) => t.id !== id),
});

export function visibleTodos({ todos, filter }: AppState): Todo[] {
  const predicates: Record<Filter, (t: Todo) => boolean> = {
    all: () => true,
    active: (t) => !t.done,
    done: (t) => t.done,
  };
  return todos.filter(predicates[filter]);
}

create table public.todos (
  id          bigint generated always as identity primary key,
  user_id     uuid not null references auth.users (id) on delete cascade default auth.uid(),
  text        text not null check (char_length(text) between 1 and 200),
  done        boolean not null default false,
  created_at  timestamptz not null default now()
);

alter table public.todos enable row level security;

create policy "본인 할 일만 조회" on public.todos
  for select using (auth.uid() = user_id);
create policy "본인 할 일만 추가" on public.todos
  for insert with check (auth.uid() = user_id);
create policy "본인 할 일만 수정" on public.todos
  for update using (auth.uid() = user_id);
create policy "본인 할 일만 삭제" on public.todos

#!/usr/bin/env python3
"""간단한 할 일 목록 CLI."""
import argparse
import json
import sys
from pathlib import Path

DB_PATH = Path.home() / ".todos.json"


def load() -> list[dict]:
    if not DB_PATH.exists():
        return []
    try:
        return json.loads(DB_PATH.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        print(f"경고: {DB_PATH} 파일이 손상되어 빈 목록으로 시작합니다.", file=sys.stderr)
        return []


def save(todos: list[dict], dry_run: bool) -> None:
    if dry_run:
        print("[dry-run] 저장하지 않았습니다.")
        return
    DB_PATH.write_text(json.dumps(todos, ensure_ascii=False, indent=2), encoding="utf-8")


def get_index(todos: list[dict], number: int) -> int:
    if not 1 <= number <= len(todos):
        sys.exit(f"오류: {number}번 할 일이 없습니다. (1~{len(todos)})")
    return number - 1


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="할 일 목록 관리")
    parser.add_argument("--dry-run", action="store_true", help="변경 내용을 저장하지 않고 미리보기")
    sub = parser.add_subparsers(dest="command", required=True)

    p_add = sub.add_parser("add", help="할 일 추가")
    p_add.add_argument("text", help="할 일 내용")
    sub.add_parser("list", help="목록 보기")
    for name, help_text in (("done", "완료 처리"), ("remove", "삭제")):
        p = sub.add_parser(name, help=help_text)
        p.add_argument("number", type=int, help="list에 표시된 번호")

    args = parser.parse_args(argv)
    todos = load()

    if args.command == "add":
        todos.append({"text": args.text, "done": False})
        print(f"추가: {args.text}")
        save(todos, args.dry_run)
    elif args.command == "list":
        if not todos:
            print("할 일이 없습니다.")
        for i, t in enumerate(todos, 1):
            print(f"{i:>3}. [{'x' if t['done'] else ' '}] {t['text']}")
    elif args.command == "done":
        i = get_index(todos, args.number)
        todos[i]["done"] = True
        print(f"완료: {todos[i]['text']}")
        save(todos, args.dry_run)
    elif args.command == "remove":
        i = get_index(todos, args.number)
        removed = todos.pop(i)
        print(f"삭제: {removed['text']}")
        save(todos, args.dry_run)
    return 0


if __name__ == "__main__":
    sys.exit(main())

@@prompt
할 일 목록 앱을 만들어줘. 항목이 나타나고 사라지고 순서가 바뀔 때 Motion 애니메이션 라이브러리(motion/react)로 스프링 전환을 넣어줘.
@@transcript
`motion` 패키지를 설치하고(`npm install motion`), 리스트 항목에 진입/퇴장/재정렬 애니메이션을 붙였습니다.

- `AnimatePresence` — 삭제되는 항목이 바로 사라지지 않고 퇴장 애니메이션을 재생
- `layout` 속성 — 완료된 항목이 아래로 정렬될 때 위치가 부드럽게 이동
- `transition={{ type: "spring" }}` — 딱딱한 easing 대신 스프링 물리
@@file src/App.tsx
import { useState } from "react";
import { AnimatePresence, motion } from "motion/react";

type Todo = { id: number; text: string; done: boolean };
const spring = { type: "spring", stiffness: 500, damping: 35 } as const;

export default function App() {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [text, setText] = useState("");

  // 완료된 항목은 아래로 → layout 애니메이션으로 이동
  const sorted = [...todos].sort((a, b) => Number(a.done) - Number(b.done));

  function add(e: React.FormEvent) {
    e.preventDefault();
    if (!text.trim()) return;
    setTodos((p) => [{ id: Date.now(), text: text.trim(), done: false }, ...p]);
    setText("");
  }

  return (
    <main className="app">
      <h1>할 일 목록</h1>
      <form onSubmit={add}>
        <input value={text} onChange={(e) => setText(e.target.value)} placeholder="할 일을 입력하세요" />
        <motion.button whileTap={{ scale: 0.92 }} whileHover={{ scale: 1.04 }}>추가</motion.button>
      </form>
      <ul>
        <AnimatePresence initial={false}>
          {sorted.map((t) => (
            <motion.li
              key={t.id}
              layout
              initial={{ opacity: 0, y: -12, scale: 0.96 }}
              animate={{ opacity: t.done ? 0.5 : 1, y: 0, scale: 1 }}
              exit={{ opacity: 0, x: 40, transition: { duration: 0.18 } }}
              transition={spring}
            >
              <input
                type="checkbox"
                checked={t.done}
                onChange={() => setTodos((p) => p.map((x) => (x.id === t.id ? { ...x, done: !x.done } : x)))}
              />
              <span style={{ textDecoration: t.done ? "line-through" : "none" }}>{t.text}</span>
              <button onClick={() => setTodos((p) => p.filter((x) => x.id !== t.id))}>삭제</button>
            </motion.li>
          ))}
        </AnimatePresence>
      </ul>
    </main>
  );
}
