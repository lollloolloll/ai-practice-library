@@code
// src/App.jsx
import { useEffect, useState } from 'react';
import Header from './components/Header';
import TodoInput from './components/TodoInput';
import FilterBar from './components/FilterBar';
import TodoList from './components/TodoList';

export default function App() {
  const [todos, setTodos] = useState(() => JSON.parse(localStorage.getItem('todos') || '[]'));
  const [filter, setFilter] = useState('all');
  useEffect(() => localStorage.setItem('todos', JSON.stringify(todos)), [todos]);

  const add = (text) => setTodos((p) => [...p, { id: Date.now(), text, done: false }]);
  const toggle = (id) => setTodos((p) => p.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));
  const remove = (id) => setTodos((p) => p.filter((t) => t.id !== id));
  const visible = todos.filter((t) => (filter === 'all' ? true : filter === 'done' ? t.done : !t.done));

  return (
    <div className="app">
      <Header left={todos.filter((t) => !t.done).length} />
      <TodoInput onAdd={add} />
      <FilterBar value={filter} onChange={setFilter} />
      <TodoList todos={visible} onToggle={toggle} onRemove={remove} />
    </div>
  );
}

let todos = [];
const listeners = new Set();

export const getTodos = () => todos;

export function setTodos(next) {
  todos = next;
  listeners.forEach((fn) => fn(todos));
}

export function subscribe(fn) {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

- `todo-logic.js` — 순수 함수 (입력 → 새 배열)
- `todo-storage.js` — 저장소 어댑터 (`load`/`save` 인터페이스만 지키면 교체 가능)
- `todo-view.js` — DOM 렌더링과 사용자 입력 → 콜백 호출
- `app.js` — 세 계층을 연결

// 비즈니스 규칙만. DOM·저장소 의존성 없음.
export const MAX_LENGTH = 100;

export function createTodo(todos, rawText) {
  const text = rawText.trim();
  if (!text) return { todos, error: '내용을 입력하세요.' };
  if (text.length > MAX_LENGTH) return { todos, error: `${MAX_LENGTH}자 이하로 입력하세요.` };
  return { todos: [...todos, { id: crypto.randomUUID(), text, done: false }], error: null };
}

export const toggleTodo = (todos, id) => todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
export const deleteTodo = (todos, id) => todos.filter((t) => t.id !== id);
export const countRemaining = (todos) => todos.filter((t) => !t.done).length;

// 화면 책임만. 데이터를 받아 그리고, 사용자 의도를 콜백으로 알린다.
export function createView(root, { onAdd, onToggle, onDelete }) {
  const form = root.querySelector('form');
  const input = root.querySelector('input[name=text]');
  const list = root.querySelector('ul');
  const status = root.querySelector('.status');

  form.addEventListener('submit', (e) => {
    e.preventDefault();
    onAdd(input.value);
  });

  return {
    render(todos, remaining, error) {
      list.replaceChildren(...todos.map((t) => {
        const li = document.createElement('li');
        li.className = t.done ? 'done' : '';
        const box = Object.assign(document.createElement('input'), { type: 'checkbox', checked: t.done });
        box.onchange = () => onToggle(t.id);
        const span = Object.assign(document.createElement('span'), { textContent: t.text });
        const del = Object.assign(document.createElement('button'), { textContent: '삭제' });
        del.onclick = () => onDelete(t.id);
        li.append(box, span, del);
        return li;
      }));
      status.textContent = error ?? `남은 할 일 ${remaining}개`;
      status.classList.toggle('error', Boolean(error));
      if (!error) input.value = '';
    },
  };
}

MVC 세 클래스로 나눴습니다. View와 Model은 서로를 전혀 모르고, Controller만 둘을 압니다.

- **Model** — 할 일 배열 보관, 변경 시 `onChange` 콜백 호출, localStorage 저장
- **View** — DOM 생성과 이벤트 바인딩 메서드(`bindAdd`, `bindToggle`, `bindDelete`) 제공
- **Controller** — View 이벤트 → Model 메서드 호출, Model 변경 → View 렌더
@@file js/model.js
export class TodoModel {
  constructor() {
    this.todos = JSON.parse(localStorage.getItem('todos') || '[]');
    this.onChange = () => {};
  }
  #commit() {
    localStorage.setItem('todos', JSON.stringify(this.todos));
    this.onChange(this.todos);
  }
  add(text) {
    this.todos.push({ id: Date.now(), text, done: false });
    this.#commit();
  }
  toggle(id) {
    this.todos = this.todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
    this.#commit();
  }
  remove(id) {
    this.todos = this.todos.filter((t) => t.id !== id);
    this.#commit();
  }
}

export class TodoView {
  constructor(root) {
    this.form = root.querySelector('form');
    this.input = root.querySelector('input');
    this.list = root.querySelector('ul');
    this.count = root.querySelector('.count');

  render(todos) {
    this.list.innerHTML = '';
    for (const t of todos) {
      const li = document.createElement('li');
      li.dataset.id = t.id;
      li.className = t.done ? 'done' : '';
      li.innerHTML = '<input type="checkbox"><span></span><button>삭제</button>';
      li.querySelector('input').checked = t.done;
      li.querySelector('span').textContent = t.text;
      this.list.append(li);

    this.count.textContent = `남은 할 일 ${todos.filter((t) => !t.done).length}개`;
  }
  bindAdd(handler) {
    this.form.addEventListener('submit', (e) => {
      e.preventDefault();
      const text = this.input.value.trim();
      if (text) { handler(text); this.input.value = ''; }
    });
  }
  bindToggle(handler) {
    this.list.addEventListener('change', (e) => handler(Number(e.target.closest('li').dataset.id)));

export class TodoController {
  constructor(model, view) {
    this.model = model;
    this.view = view;

    view.bindToggle((id) => model.toggle(id));
    view.bindDelete((id) => model.remove(id));

}

모든 상태 변경을 `todoReducer` 한 곳에 모았습니다. 컴포넌트는 `useTodos()`로 상태를 읽고 `dispatch({ type })`로 "무슨 일이 일어났는지"만 알립니다. props를 여러 단계로 내려보내는 일(prop drilling)이 사라지고, 상태가 어떻게 바뀌는지는 reducer만 보면 됩니다.

const { createContext, useContext, useReducer, useEffect, useState } = React;

// ── 스토어: 상태 변경은 오직 여기서 ──
const initialState = { todos: JSON.parse(localStorage.getItem('todos') || '[]'), filter: 'all' };

function todoReducer(state, action) {
  switch (action.type) {
    case 'added':
      return { ...state, todos: [...state.todos, { id: Date.now(), text: action.text, done: false }] };
    case 'toggled':
      return { ...state, todos: state.todos.map((t) => (t.id === action.id ? { ...t, done: !t.done } : t)) };
    case 'deleted':
      return { ...state, todos: state.todos.filter((t) => t.id !== action.id) };
    case 'clearedCompleted':
      return { ...state, todos: state.todos.filter((t) => !t.done) };
    case 'filterChanged':
      return { ...state, filter: action.filter };
    default:
      throw new Error('알 수 없는 액션: ' + action.type);
  }
}

const TodoContext = createContext(null);

function TodoProvider({ children }) {
  const [state, dispatch] = useReducer(todoReducer, initialState);
  useEffect(() => localStorage.setItem('todos', JSON.stringify(state.todos)), [state.todos]);
  return <TodoContext.Provider value={{ state, dispatch }}>{children}</TodoContext.Provider>;
}
const useTodos = () => useContext(TodoContext);

const { useState, useEffect, useRef, useCallback } = React;

// hooks/useLocalStorage.js
function useLocalStorage(key, initialValue) {
  const [value, setValue] = useState(() => {
    try {
      const stored = localStorage.getItem(key);
      return stored !== null ? JSON.parse(stored) : initialValue;
    } catch {
      return initialValue;
    }
  });
  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(value));
  }, [key, value]);
  return [value, setValue];
}

// hooks/useKeyboardShortcut.js
function useKeyboardShortcut(key, handler) {
  const saved = useRef(handler);
  saved.current = handler;
  useEffect(() => {
    const onKey = (e) => {
      if (e.key === key && !['INPUT', 'TEXTAREA'].includes(document.activeElement.tagName)) {
        e.preventDefault();
        saved.current(e);
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [key]);
}

// hooks/useTodos.js
function useTodos() {
  const [todos, setTodos] = useLocalStorage('todos', []);
  const add = useCallback((text) => setTodos((p) => [...p, { id: Date.now(), text, done: false }]), [setTodos]);
  const toggle = useCallback((id) => setTodos((p) => p.map((t) => (t.id === id ? { ...t, done: !t.done } : t))), [setTodos]);
  const remove = useCallback((id) => setTodos((p) => p.filter((t) => t.id !== id)), [setTodos]);
  const remaining = todos.filter((t) => !t.done).length;
  return { todos, add, toggle, remove, remaining };
}

// App.jsx — 화면만
function App() {
  const { todos, add, toggle, remove, remaining } = useTodos();
  const [text, setText] = useState('');
  const inputRef = useRef(null);
  useKeyboardShortcut('/', () => inputRef.current?.focus());

  const submit = (e) => {
    e.preventDefault();
    if (!text.trim()) return;
    add(text.trim());
    setText('');
  };

  return (
    <div className="app">
      <h1>할 일 목록</h1>
      <form onSubmit={submit}>
        <input ref={inputRef} value={text} onChange={(e) => setText(e.target.value)} placeholder="할 일 입력 ( / 키로 포커스)" />
        <button type="submit">추가</button>
      </form>
      <ul>
        {todos.map((t) => (
          <li key={t.id} className={t.done ? 'done' : ''}>
            <input type="checkbox" checked={t.done} onChange={() => toggle(t.id)} />
            <span>{t.text}</span>
            <button onClick={() => remove(t.id)}>삭제</button>
          </li>
        ))}
      </ul>
      <p className="count">남은 할 일 {remaining}개</p>
    </div>
  );
}

const BASE = import.meta.env.VITE_API_URL ?? "http://localhost:3001";

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(BASE + path, { headers: { "Content-Type": "application/json" }, ...init });
  if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
  return res.status === 204 ? (undefined as T) : res.json();
}

|---|---|

@@term event-driven bouncing-ball-game html
@@prompt
브라우저에서 공이 튀는 게임을 만들어줘. 이벤트 기반 구조(event-driven, pub-sub)로, 게임이 'bounce'·'score'·'gameover' 이벤트를 내보내면 UI와 효과음이 구독하게 해줘.
@@transcript
게임 로직은 화면 UI나 소리를 **직접 호출하지 않고** 이벤트만 발행합니다. 구독자는 서로를 모릅니다.

```
Game ──emit──▶ bus ──▶ HUD(점수 표시)
                  ├──▶ Sound(효과음)
                  └──▶ Stats(최고 점수 저장)
```

새 기능(예: 진동, 파티클)을 추가할 때 게임 코드를 건드리지 않고 `bus.on('bounce', ...)` 한 줄로 붙일 수 있습니다.
@@code

// ── 이벤트 버스 ──
const bus = {
  handlers: {},
  on(type, fn) { (this.handlers[type] ||= []).push(fn); },
  emit(type, payload) { (this.handlers[type] || []).forEach((fn) => fn(payload)); },
};

// ── 게임: 상태 계산 + 이벤트 발행만 ──
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');
const game = { ball: null, paddleX: 210, score: 0, running: false };

function start() {
  Object.assign(game, { ball: { x: 210, y: 100, vx: 3, vy: 0, r: 11 }, score: 0, running: true });
  bus.emit('start');
}

function step() {
  const b = game.ball;
  if (game.running) {
    b.vy += 0.3; b.x += b.vx; b.y += b.vy;
    if (b.x < b.r || b.x > canvas.width - b.r) { b.vx *= -1; b.x = Math.max(b.r, Math.min(canvas.width - b.r, b.x)); bus.emit('bounce', { kind: 'wall' }); }
    const py = canvas.height - 30;
    if (b.vy > 0 && b.y + b.r >= py && b.y < py + 12 && Math.abs(b.x - game.paddleX) < 50 + b.r) {
      b.y = py - b.r; b.vy = -11; b.vx += (b.x - game.paddleX) * 0.06;
      game.score++;
      bus.emit('bounce', { kind: 'paddle' });
      bus.emit('score', game.score);
    }
    if (b.y > canvas.height + 20) { game.running = false; bus.emit('gameover', game.score); }
  }
  draw();
  requestAnimationFrame(step);
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = '#e2e8f0';
  ctx.fillRect(game.paddleX - 50, canvas.height - 30, 100, 12);
  if (game.ball) {
    ctx.fillStyle = '#f97316';
    ctx.beginPath(); ctx.arc(game.ball.x, game.ball.y, game.ball.r, 0, Math.PI * 2); ctx.fill();
  }
}

canvas.addEventListener('pointermove', (e) => {
  const rect = canvas.getBoundingClientRect();
  game.paddleX = ((e.clientX - rect.left) / rect.width) * canvas.width;
});
canvas.addEventListener('pointerdown', () => { if (!game.running) start(); });

// ── 구독자 1: HUD ──
const $ = (id) => document.getElementById(id);
bus.on('start', () => { $('score').textContent = 0; $('msg').textContent = '패들로 공을 받아내세요!'; });
bus.on('score', (s) => { $('score').textContent = s; });
bus.on('gameover', (s) => { $('msg').textContent = `게임 오버 — ${s}점. 클릭해서 다시 시작`; });

// ── 구독자 2: 효과음 (Web Audio) ──
let audio;
bus.on('start', () => { audio ||= new AudioContext(); });
bus.on('bounce', ({ kind }) => {
  if (!audio) return;
  const osc = audio.createOscillator(), gain = audio.createGain();
  osc.frequency.value = kind === 'paddle' ? 660 : 330;
  gain.gain.setValueAtTime(0.15, audio.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audio.currentTime + 0.12);
  osc.connect(gain).connect(audio.destination);
  osc.start(); osc.stop(audio.currentTime + 0.12);
});

// ── 구독자 3: 최고 점수 저장 ──
let best = Number(localStorage.getItem('best') || 0);
$('best').textContent = best;
bus.on('gameover', (s) => {
  if (s > best) { best = s; localStorage.setItem('best', best); $('best').textContent = best; }
});

step();
</script>
</body>
